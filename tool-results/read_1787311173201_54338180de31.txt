   384→/* ------------------------------------------------------------------ */
   385→/*  Main Component                                                     */
   386→/* ------------------------------------------------------------------ */
   387→
   388→export default function ConfiguratorView() {
   389→  const {
   390→    configurator,
   391→    viewParams,
   392→    isAuthenticated,
   393→    setConfiguratorStep,
   394→    setConfiguratorResponse,
   395→    setConfiguratorData,
   396→    resetConfigurator,
   397→    requireAuth,
   398→    navigate,
   399→    addToast,
   400→  } = useAppStore();
   401→
   402→  const responses = configurator.responses;
   403→
   404→  // Use local step ID for reliable navigation across conditional step changes
   405→  const [localStepId, setLocalStepId] = useState<string>('project-type');
   406→  const [direction, setDirection] = useState(0);
   407→  const [isSubmitting, setIsSubmitting] = useState(false);
   408→  const [referenceNumber, setReferenceNumber] = useState('');
   409→  const isInitialized = useRef(false);
   410→
   411→  // Build visible steps from responses
   412→  const steps = useMemo(() => buildSteps(responses), [responses]);
   413→
   414→  // Find current index from step ID — falls back to 0 if ID not found
   415→  const currentIdx = useMemo(
   416→    () => Math.max(0, steps.findIndex(s => s.id === localStepId)),
   417→    [steps, localStepId]
   418→  );
   419→  const activeStep = steps[currentIdx] || steps[0];
   420→
   421→  // Progress (exclude confirmation from progress calc)
   422→  const progressSteps = steps.filter(s => s.type !== 'confirmation');
   423→  const progressIdx = progressSteps.findIndex(s => s.id === localStepId);
   424→  const progressPercent = progressSteps.length > 1
   425→    ? ((Math.max(0, progressIdx) + 1) / progressSteps.length) * 100
   426→    : 0;
   427→
   428→  // Sync step index to store
   429→  useEffect(() => {
   430→    setConfiguratorStep(currentIdx);
   431→  }, [currentIdx, setConfiguratorStep]);
   432→
   433→  // Pre-select model from viewParams on first mount
   434→  useEffect(() => {
   435→    if (isInitialized.current) return;
   436→    isInitialized.current = true;
   437→
   438→    if (viewParams.modelId && !responses.projectType) {
   439→      const model = useAppStore.getState().selectedModel;
   440→      if (model) {
   441→        const cat = model.categoryName?.toLowerCase() || '';
   442→        const typeMap: Record<string, string> = {
   443→          villa: 'villa', duplex: 'duplex', triplex: 'triplex', immeuble: 'immeuble',
   444→        };
   445→        const mapped = Object.entries(typeMap).find(([k]) => cat.includes(k))?.[1] || 'villa';
   446→        setConfiguratorResponse('projectType', mapped);
   447→        setConfiguratorData({ modelId: model.id, categoryName: model.categoryName });
   448→      }
   449→    }
   450→  // eslint-disable-next-line react-hooks/exhaustive-deps
   451→  }, []);
   452→
   453→  /* Navigate helpers */
   454→  const goToStepId = useCallback((stepId: string, dir: number) => {
   455→    const exists = steps.some(s => s.id === stepId);
   456→    if (!exists) return;
   457→    setDirection(dir);
   458→    setLocalStepId(stepId);
   459→  }, [steps]);
   460→
   461→  const goNext = useCallback(() => {
   462→    const nextIdx = currentIdx + 1;
   463→    if (nextIdx < steps.length) {
   464→      setDirection(1);
   465→      setLocalStepId(steps[nextIdx].id);
   466→    }
   467→  }, [currentIdx, steps]);
   468→
   469→  const goBack = useCallback(() => {
   470→    const prevIdx = currentIdx - 1;
   471→    if (prevIdx >= 0) {
   472→      setDirection(-1);
   473→      setLocalStepId(steps[prevIdx].id);
   474→    }
   475→  }, [currentIdx, steps]);
   476→
   477→  const handleSkip = useCallback(() => goNext(), [goNext]);
   478→
   479→  /* Submit project */
   480→  const handleSubmit = useCallback(async () => {
   481→    setIsSubmitting(true);
   482→    try {
   483→      const ref = generateReference();
   484→      setReferenceNumber(ref);
   485→
   486→      const payload = {
   487→        ...responses,
   488→        referenceNumber: ref,
   489→        status: 'draft',
   490→        createdAt: new Date().toISOString(),
   491→      };
   492→
   493→      const res = await fetch('/api/projects', {
   494→        method: 'POST',
   495→        headers: { 'Content-Type': 'application/json' },
   496→        body: JSON.stringify(payload),
   497→      });
   498→
   499→      if (!res.ok) throw new Error('Erreur serveur');
   500→
   501→      goNext();
   502→      addToast('Projet créé avec succès !', 'success');
   503→    } catch {
   504→      // Allow progression for demo / offline
   505→      const ref = generateReference();
   506→      setReferenceNumber(ref);
   507→      goNext();
   508→      addToast('Projet enregistré !', 'success');
   509→    } finally {
   510→      setIsSubmitting(false);
   511→    }
   512→  }, [responses, goNext, addToast]);
   513→
   514→  const handleSummarySubmit = useCallback(() => {
   515→    if (!isAuthenticated) {
   516→      requireAuth('projects');
   517→    } else {
   518→      handleSubmit();
   519→    }
   520→  }, [isAuthenticated, requireAuth, handleSubmit]);
   521→
   522→  const handleFinalAction = useCallback(() => {
   523→    resetConfigurator();
   524→    navigate('projects');
   525→  }, [resetConfigurator, navigate]);
   526→
   527→  /* Can we proceed to next? */
   528→  const canProceed = useMemo(() => {
   529→    if (!activeStep) return false;
   530→    if (activeStep.type === 'summary' || activeStep.type === 'confirmation') return true;
   531→    if (activeStep.type === 'textarea' || activeStep.type === 'slider' || activeStep.type === 'counter') return true;
   532→    if (activeStep.type === 'choice-multi') return true;
   533→    if (activeStep.skippable) return true;
   534→    const val = responses[activeStep.responseKey];
   535→    return val !== undefined && val !== null && val !== '';
   536→  }, [activeStep, responses]);
   537→
   538→  /* ---------------------------------------------------------------- */
   539→/*  Step renderers                                                   */
   540→/* ---------------------------------------------------------------- */
   541→
   542→  const renderChoiceSingle = (step: StepDef) => {
   543→    const selected = responses[step.responseKey] as string | undefined;
   544→    const isGrid = (step.options?.length ?? 0) > 4;
   545→
   546→    return (
   547→      <div className={
   548→        isGrid
   549→          ? 'grid grid-cols-2 sm:grid-cols-3 gap-3'
   550→          : 'grid grid-cols-1 gap-2.5'
   551→      }>
   552→        {step.options?.map(opt => {
   553→          const isSelected = selected === opt.value;
   554→          const Icon = opt.icon;
   555→          return (
   556→            <motion.button
   557→              key={opt.value}
   558→              type="button"
   559→              whileTap={{ scale: 0.96 }}
   560→              onClick={() => {
   561→                setConfiguratorResponse(step.responseKey, opt.value);
   562→                // Auto-advance on non-first, non-skippable single choice steps
   563→                if (step.id !== 'project-type' && !step.skippable) {
   564→                  setTimeout(() => {
